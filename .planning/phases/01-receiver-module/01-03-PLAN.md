**Start(ctx context.Context, host component.Host) error:**
1. Store host: `r.host = host`
2. Create gRPC connection: `conn, err := r.cfg.ClientConfig.ToClientConn(ctx, host, r.settings.TelemetrySettings)`. Note: check the actual API signature for `ToClientConn` — per research it takes `(ctx, host.GetExtensions(), telemetrySettings)` or similar. Use the correct call signature. If it does not accept host directly, use the pattern from Research Pattern 3.
3. On error, return `fmt.Errorf("failed to create gRPC client connection: %w", err)`
4. Store connection: `r.conn = conn`
5. Create Tetragon client: `r.client = tetragonv1.NewFineGuidanceSensorsClient(conn)`
6. Create stream context: `streamCtx, cancel := context.WithCancel(context.Background())` — NEVER pass Start's ctx (per STATE.md Pitfall 1)
7. Store cancel: `r.cancel = cancel`
8. Start goroutine: `r.wg.Add(1); go r.streamEvents(streamCtx)`
9. Return nil

**Shutdown(ctx context.Context) error:**
1. `r.cancel()` — safe because initialized to no-op in factory (per STATE.md Pitfall 2)
2. `r.wg.Wait()` — wait for goroutine exit
3. If `r.conn != nil`: return `r.conn.Close()`
4. Return nil

**streamEvents(ctx context.Context):**
```go
func (r *tetragonReceiver) streamEvents(ctx context.Context) {
    defer r.wg.Done()

    eventCh := make(chan *tetragonv1.GetEventsResponse, 1000)

    // Start consumer goroutine
    var consumeWg sync.WaitGroup
    consumeWg.Add(1)
    go func() {
        defer consumeWg.Done()
        r.consumeChannel(ctx, eventCh)
    }()

    b := backoff.NewExponentialBackOff()
    b.InitialInterval = r.cfg.Retry.InitialInterval
    b.MaxInterval = r.cfg.Retry.MaxInterval
    b.MaxElapsedTime = r.cfg.Retry.MaxElapsedTime

    for {
        if ctx.Err() != nil {
            close(eventCh)
            consumeWg.Wait()
            return
        }
        err := r.runStream(ctx, eventCh)
        if ctx.Err() != nil {
            close(eventCh)
            consumeWg.Wait()
            return
        }
        // Transient error — report and retry
        componentstatus.ReportStatus(r.host,
            componentstatus.NewRecoverableErrorEvent(err))
        r.logger.Warn("stream error, reconnecting",
            zap.Error(err),
            zap.Duration("backoff", b.NextBackOff()))

        wait := b.NextBackOff()
        if wait == backoff.Stop {
            r.logger.Error("max backoff elapsed, stopping stream")
            close(eventCh)
            consumeWg.Wait()
            return
        }
        select {
        case <-ctx.Done():
            close(eventCh)
            consumeWg.Wait()
            return
        case <-time.After(wait):
        }
    }
}
```

**runStream(ctx context.Context, eventCh chan<- *tetragonv1.GetEventsResponse) error:**
```go
func (r *tetragonReceiver) runStream(ctx context.Context, eventCh chan<- *tetragonv1.GetEventsResponse) error {
    stream, err := r.client.GetEvents(ctx, &tetragonv1.GetEventsRequest{})
    if err != nil {
        return fmt.Errorf("failed to open event stream: %w", err)
    }

    r.logger.Info("connected to Tetragon, streaming events")
    componentstatus.ReportStatus(r.host,
        componentstatus.NewEvent(componentstatus.StatusOK))

    // Reset backoff on successful connect
    for {
        resp, err := stream.Recv()
        if err != nil {
            if ctx.Err() != nil {
                return ctx.Err()
            }
            return fmt.Errorf("stream recv error: %w", err)
        }
        select {
        case eventCh <- resp:
        case <-ctx.Done():
            return ctx.Err()
        }
    }
}
```

```go
func (r *tetragonReceiver) consumeChannel(ctx context.Context, eventCh <-chan *tetragonv1.GetEventsResponse) {
    for resp := range eventCh {
        logs := convertEvent(resp)
        obsCtx := r.obsReport.StartLogsOp(ctx)
        err := r.consumer.ConsumeLogs(obsCtx, logs)
        r.obsReport.EndLogsOp(obsCtx, "tetragon", logs.LogRecordCount(), err)
        if err != nil {
            r.logger.Error("failed to consume logs", zap.Error(err))
        }
    }
}
```

```go
// mockGetEventsClient implements tetragonv1.FineGuidanceSensors_GetEventsClient
type mockGetEventsClient struct {
    grpc.ClientStream
    responses []*tetragonv1.GetEventsResponse
    idx       int
    err       error
    mu        sync.Mutex
}

func (m *mockGetEventsClient) Recv() (*tetragonv1.GetEventsResponse, error) {
    m.mu.Lock()
    defer m.mu.Unlock()
    if m.err != nil {
        return nil, m.err
    }
    if m.idx >= len(m.responses) {
        // Block until context cancelled (simulate long-running stream)
        m.mu.Unlock()
        <-time.After(10 * time.Second)
        m.mu.Lock()
        return nil, io.EOF
    }
    resp := m.responses[m.idx]
    m.idx++
    return resp, nil
}

// mockTetragonClient implements tetragonClient
type mockTetragonClient struct {
    mu           sync.Mutex
    stream       *mockGetEventsClient
    getEventsErr error
    callCount    int
}

func (m *mockTetragonClient) GetEvents(ctx context.Context, in *tetragonv1.GetEventsRequest, opts ...grpc.CallOption) (
    tetragonv1.FineGuidanceSensors_GetEventsClient, error) {
    m.mu.Lock()
    defer m.mu.Unlock()
    m.callCount++
    if m.getEventsErr != nil {
        return nil, m.getEventsErr
    }
    return m.stream, nil
}
```
